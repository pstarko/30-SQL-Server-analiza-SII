/*
1. Pokaż jak kształtowała się liczba zatrudnien w poszczególnych latach. W dodatkowej
kolumnie dodaj narastające wartości ilości zatrudnień w sumie dla każdego roku

2. Zmień powyższy raport o informacje na jaki etat zostali zatrudnieni pracownicy. Również
dodaj kolumnę z wartościami narastającymi(47 wierszy)


*/
--1
select
	year(hire_date) rok,
	count(employee_id) ilosc,
	sum(count(employee_id)) over (order by year(hire_date)) suma
	from employees
group by 
	year(hire_date)

--2
select
	year(hire_date) rok,
	j.job_title,
	count(employee_id) ilosc,
	sum(count(employee_id)) over (order by year(hire_date), j.job_title) suma
	from employees e join jobs j on e.job_id = j.job_id
group by 
	year(hire_date), 
	j.job_title

/*
3. Posortuj dane pracowników po job_title. Pokaż first and last name oraz salary. W dodatkowej
kolumnie pokaż ile w sumie wydajemy dla całego etatu. Kolejna kolumna to procent udziału
poszczególnego pracownika w wydatkach na etat


*/
--3
select
	t1.*,
	CONCAT(CAST(t1.salary / t1.suma_jobs as money) * 100, '%') prct
	from (
		select
			j.job_title,
			e.first_name,
			e.last_name,
			e.salary,
			sum(e.salary) over (order by j.job_title) suma_jobs
			from employees e join jobs j on e.job_id = j.job_id) t1

/*
5. Pokaż raport pracowników zatrudnionych w poszczególnych departamentach (11 wierszy)

6. Zaktualizuj powyższe zapytanie w taki sposób, by otrzymać tylko pierwsze dwa nazwiska

*/
--5
SELECT
    d.department_name,
    STRING_AGG(last_name, ',') lista
FROM
    employees e join departments d on e.department_id=d.department_id
group by
    d.department_name


--6
select
    t1.department_name,
    STRING_AGG(last_name, ',') lista
    from (
    SELECT
        d.department_name,
        last_name,
        rank() over (partition by d.department_name order by last_name) rnk
    FROM
        employees e join departments d on e.department_id=d.department_id ) t1
where
    t1.rnk <= 2
group by
    t1.department_name

/*
8. Przeanalizuj zarobki w osobnych raportach
a. Wzrost wynagrodzeń w kolejnych kwartałach(25 wierszy)

b. Które kwartały bez względu na rok generowały większe zatrudnienie, utwórz ranking
i pokaż również jaka jest różnica zatrudnień pomiędzy kwartałami 1 i 2, 2 i 3, 3 i 4
*/

    DATEPART(QUARTER, hire_date) kwartal,
    sum(sum(Salary)) over (order by year(hire_date), DATEPART(QUARTER, hire_date)) wzrost
    from employees e
group by
    year(hire_date),
    DATEPART(QUARTER, hire_date)


select
    t1.*,
    t1.poprzednik - t1.ilosc roznica,
    rank() over (order by t1.ilosc desc) rnk
    from (
        select 
            DATEPART(QUARTER, hire_date) kwartal,
            count(employee_id) ilosc,

            from employees e

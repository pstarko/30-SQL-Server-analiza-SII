--warunkowe

SELECT
    last_name,
    salary,
    CASE
        WHEN salary > 10000 THEN 'OK'
        ELSE'---'
    END AS warunek
FROM
    employees

SELECT
    last_name,
    salary,
    CASE
        WHEN salary > 15000 THEN '>15k'
        WHEN salary > 10000 THEN '>10k'
        ELSE'---'
    END AS warunek
FROM
    employees

SELECT
    last_name,
    salary,
    job_id,
    CASE
        WHEN salary > 15000 THEN '>15k'
        WHEN salary > 10000 and job_id = 'FI_MGR' THEN '>10k'
        ELSE'---'
    END AS warunek
FROM
    employees

SELECT
    hire_date,
    year(hire_date) rok,
    month(hire_date) miesiac,
    day(hire_date) dzien
from employees

SELECT
    hire_date,
    getdate(),
    DATEDIFF(year, hire_date, getdate()) result_year,
    DATEDIFF(month, hire_date, getdate()) result_month,
    DATEDIFF(day, hire_date, getdate()) result_day
from employees


SELECT
    hire_date,
    DATEADD(year, 2, hire_date) result_year,
    DATEADD(month, 2,hire_date) result_month,
    DATEADD(day, 2, hire_date) result_day,
    DATEADD(day, -2, hire_date) result_day
from employees

/*
1. Pokaż imię, nazwisko, numer działu i stanowisko dla wszystkich pracowników (106
wierszy)
*/

select
	e.first_name,
	e.last_name,
	e.department_id,
	j.job_title
	from employees e join jobs j on e.job_id = j.job_id
where
	e.department_id is not null

/*
2. Pokaż pracowników pracujących w Londynie i Toronto(3 wiersze)

*/

select
	e.first_name,
	e.last_name,
	l.city
	from employees e join departments d on e.department_id = d.department_id
					 join locations l on d.location_id = l.location_id
where
	l.city in ('London','Toronto')

select
	e.first_name,
	e.last_name,
	l.city
	from employees e join departments d on e.department_id = d.department_id
					 join locations l on d.location_id = l.location_id and l.city in ('London','Toronto')


/*
3. Pokaż poniższy raport nazwisk i nazw departamentów. Uwzględnij departamenty,
które nie zatrudniają pracowników. Nie bierz pod uwagę zespołu Operations(121
wierszy)

*/

select
	e.last_name,
	d.department_name
	from employees e right join departments d on e.department_id = d.department_id
where
	d.department_name not in ('Operations')
/*
4. Pokaż ilość zatrudnionych pracowników w Stretford, Bombay, Sydney, London,
Munich i Seattle(6 wierszy)


*/

select
	l.city,
	count(e.employee_id) ilosc_pracownikow
	from employees e join departments d on e.department_id = d.department_id
					 right join locations l on d.location_id = l.location_id
where
	l.city in ('Stretford', 'Bombay', 'Sydney', 'London','Munich', 'Seattle')
group by
	l.city
/*
5. Pokaż liczbę pracowników, którzy nie pracują w prowincji stanowej, zaczynając od
litery T (1 wiersz) (1 wiersz)


*/

select
	count(e.employee_id) ilosc_pracownikow
	from employees e join departments d on e.department_id = d.department_id
					 join locations l on d.location_id = l.location_id
where
	left(l.state_province, 1) != 'T'

/*
6. Pokaż wszystkie informacje o pracownikach, którzy są zatrudnieni na stanowisku
Accountant oraz Puchasing Clerk, których roczny dochód przekracza 37200. Roczny
dochód jest dwunastokrotnością wynagrodzenia podstawowego wraz z prowizją (5
wierszy)
*/

select
	e.first_name,
	e.last_name,
	j.job_title,
	(e.salary + (e.salary * isnull(e.commission_pct,0))) * 12 year_income
	from employees e join jobs j on e.job_id = j.job_id
where
	j.job_title in ('Accountant', 'Purchasing Clerk')
	and (e.salary + (e.salary * isnull(e.commission_pct,0))) * 12 > 37200


/*
10. Pokaż te job_title, gdzie ilość zatrudnionych na to stanowisko pracowników
przekracza liczbę 10 (3 wiersze)

*/

select
	j.job_title,
	count(e.employee_id) ilosc_zatrudnionych
	from employees e join jobs j on e.job_id = j.job_id
group by
	j.job_title
having
	count(e.employee_id) > 10

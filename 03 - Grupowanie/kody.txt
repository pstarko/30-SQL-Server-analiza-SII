select 
    department_id,
	job_id,
	grouping(department_id) grup_dept,
	grouping(job_id) grup_job_id,
    Sum(salary) suma
    from employees
where
	department_id in (10,20,30)
	or department_id is null
group by
	cube(department_id, job_id)

    department_id,
	job_id,
	grouping(department_id) grup_dept,
	grouping(job_id) grup_job_id,

    from employees
where
	department_id in (10,20,30)
	or department_id is null
group by
	cube(department_id, job_id)

	grouping(department_id) = 1

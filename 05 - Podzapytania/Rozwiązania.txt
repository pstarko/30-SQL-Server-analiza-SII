/*
1. Wyświetl nazwiska i stanowiska pracowników pracujących w tym samym dziale co
pracownik o nazwisku Yang (3 wiersze)
*/

select
	e.first_name,
	e.last_name,
	(select job_title from jobs j where e.job_id = j.job_id) jt
	from employees e
where
	e.department_id = (select department_id from employees
							where last_name = 'Yang')

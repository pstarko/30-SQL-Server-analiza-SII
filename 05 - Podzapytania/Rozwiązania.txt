/*
1. Wyświetl nazwiska i stanowiska pracowników pracujących w tym samym dziale co
pracownik o nazwisku Yang (3 wiersze)
*/

select
	e.first_name,
	e.last_name,
	(select job_title from jobs j where e.job_id = j.job_id) jt
	from employees e
where
	e.department_id = (select department_id from employees
							where last_name = 'Yang')


/*
2. Wyświetl następujące dane o najdłużej zatrudnionym pracowniku na stanowisku
FI_ACCOUNT (1 wiersz)
*/

select
	*
	from employees e
where
	hire_date = (select min(hire_date) from employees
						where job_id = 'FI_ACCOUNT')
	and job_id = 'FI_ACCOUNT'

/*
3. Zobacz najdłużej pracujących pracowników każdego działu. (12 wierszy)
*/

select
	e.first_name,
	e.last_name,
	e.department_id,
	e.hire_date
	from employees e
where
	e.hire_date = (select min(hire_date) from employees
					where department_id = e.department_id)
/*
4. Wyświetlaj te nazwy stanowisk, w których nie zatrudniono nikogo z działu
MARKETING (17 wierszy)
*/

select
	*
	from jobs
where
	job_id in (
			select
				job_id
				from employees
			where
				department_id not in (
								select
									department_id
									from departments
								where 
									department_name = 'Marketing'))
